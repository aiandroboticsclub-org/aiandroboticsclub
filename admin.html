// Configuration constants const REPO_OWNER = 'aiandroboticsclub-org'; const REPO_NAME = 'aiandroboticsclub'; const FILE_PATH = 'products.json'; // Auto-fill token from browser storage if previously saved locally document.addEventListener('DOMContentLoaded', () => { const savedToken = localStorage.getItem('gh_admin_pat'); if (savedToken) { document.getElementById('ghToken').value = savedToken; } }); document.getElementById('productForm').addEventListener('submit', async (e) => { e.preventDefault(); const tokenInput = document.getElementById('ghToken').value.trim(); if (!tokenInput) { return alert('Please enter your new GitHub Personal Access Token.'); } // Save token locally in browser storage for convenience localStorage.setItem('gh_admin_pat', tokenInput); const getUrl = `https://api.github.com/repos/${REPO_OWNER}/${REPO_NAME}/contents/${FILE_PATH}`; try { // 1. Fetch products.json from your organization repo const res = await fetch(getUrl, { headers: { 'Authorization': `Bearer ${tokenInput}`, 'Accept': 'application/vnd.github.v3+json' } }); if (!res.ok) { const err = await res.json(); throw new Error(`GitHub GET (${res.status}): ${err.message}`); } const fileData = await res.json(); const existingProducts = JSON.parse(decodeURIComponent(escape(atob(fileData.content.replace(/\s/g, ''))))); // 2. Build the new product entry const newProduct = { id: 'prod-' + Date.now(), title: document.getElementById('title').value, price: document.getElementById('price').value, image: document.getElementById('image').value || 'Images/placeholder.jpg', description: document.getElementById('description').value, featured: document.getElementById('featured').checked }; existingProducts.push(newProduct); // 3. Encode and Commit to GitHub const updatedContentBase64 = btoa(unescape(encodeURIComponent(JSON.stringify(existingProducts, null, 2)))); const putRes = await fetch(getUrl, { method: 'PUT', headers: { 'Authorization': `Bearer ${tokenInput}`, 'Content-Type': 'application/json', 'Accept': 'application/vnd.github.v3+json' }, body: JSON.stringify({ message: `Add product: ${newProduct.title}`, content: updatedContentBase64, sha: fileData.sha }) }); if (putRes.ok) { alert('Product added successfully! GitHub Pages will publish the changes in ~1-2 minutes.'); document.getElementById('productForm').reset(); // Keep saved token in input document.getElementById('ghToken').value = tokenInput; } else { const putErr = await putRes.json(); alert(`Commit Failed (${putRes.status}): ${putErr.message}`); } } catch (err) { console.error(err); alert(`Operation Failed: ${err.message}`); } });
